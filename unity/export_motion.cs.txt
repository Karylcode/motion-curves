// 匯出 Humanoid 動作給 motion-curves 網站。
// 用法：在下面 jobs 加動作、用 pick 選這次要匯出哪幾支，整段丟給 Unity MCP 的 execute_code（CodeDom，C# 6）。中文可以直接寫。
//      一次不要匯太多（約 200 格以內），execute_code 跑太久會逾時。
// 產出 public/motions/<id>/curves.json（Unity 原生曲線：關鍵影格＋切線、每格的值、Unity 實際播放方式，能在 Unity 還原一模一樣的動作）
//      public/motions/<id>/angles.json（套到 VRM 角色後每根骨頭的旋轉，每格切 samplesPerFrame 份，網頁播放用）
//      public/motions/<id>/analysis.json（關節角度、腳踩地、肌肉逐格值等給 AI 分析的數字）
// 播放方式：FBX 匯入壓縮是 Optimal 時，Unity 每條曲線各自決定怎麼播：有的是「每格一個值、格與格之間直線」，有的照動畫視窗的曲線走。
//           每條曲線的方式記在 curves.json 的 playback，整支的比對結果記在 playbackMode、playbackCheck（見下面「判斷 Unity 實際播放方式」）。
// 要匯出的動作寫在 unity/characters/<角色>.json（一個角色一個設定檔，格式見 unity/characters/HOWTO.md；_ 開頭的檔不讀）。
// MOTION_CURVES_ROOT 由 run_export.cs.txt 換成這個倉庫的資料夾。中文名、英文名、類型、描述、標籤寫在 public/motions/names.json（規範見 unity/characters/NAMING.md），改名只要改那個檔＋跑 build-index。

System.Func<Vector3, string> Vg = v => "[" + F4(-v.x) + "," + F4(v.y) + "," + F4(v.z) + "]";

  // ---------- 判斷 Unity 實際播放方式 ----------

    for (int f = 0; f <= N; f++) { AnimationUtility.SetKeyLeftTangentMode(lin, f, AnimationUtility.TangentMode.Linear); AnimationUtility.SetKeyRightTangentMode(lin, f, AnimationUtility.TangentMode.Linear); }
    denseRule[b.propertyName] = 7 * cv.length >= N;
    AnimationUtility.SetEditorCurve(tests[0], b, cv); AnimationUtility.SetEditorCurve(tests[1], b, lin); AnimationUtility.SetEditorCurve(tests[2], b, denseRule[b.propertyName] ? lin : cv);
  }
  for (int k = 0; k < 3; k++) AnimationUtility.SetAnimationClipSettings(tests[k], st);
  var cg = new GameObject[4]; var cpl = new UnityEngine.Animations.AnimationClipPlayable[4]; var cgr = new UnityEngine.Playables.PlayableGraph[4]; var cclips = new[]{ clip, tests[0], tests[1], tests[2] };
  for (int k = 0; k < 4; k++) { cg[k] = spawn(cclips[k], false); cpl[k] = lastPlayable[0]; cgr[k] = lastGraph[0]; }

    ang[a2++, f] = bend(P(HumanBodyBones.LeftUpperLeg), P(HumanBodyBones.LeftLowerLeg), P(HumanBodyBones.LeftFoot));
    ang[a2++, f] = bend(P(HumanBodyBones.RightUpperLeg), P(HumanBodyBones.RightLowerLeg), P(HumanBodyBones.RightFoot));
    ang[a2++, f] = bend(P(HumanBodyBones.LeftUpperArm), P(HumanBodyBones.LeftLowerArm), P(HumanBodyBones.LeftHand));
    ang[a2++, f] = bend(P(HumanBodyBones.RightUpperArm), P(HumanBodyBones.RightLowerArm), P(HumanBodyBones.RightHand));
    var thL = P(HumanBodyBones.LeftLowerLeg) - P(HumanBodyBones.LeftUpperLeg); var thR = P(HumanBodyBones.RightLowerLeg) - P(HumanBodyBones.RightUpperLeg);
    ang[a2++, f] = flex(thL); ang[a2++, f] = flex(thR); ang[a2++, f] = abd(thL, -1); ang[a2++, f] = abd(thR, 1);
    var uaL = P(HumanBodyBones.LeftLowerArm) - P(HumanBodyBones.LeftUpperArm); var uaR = P(HumanBodyBones.RightLowerArm) - P(HumanBodyBones.RightUpperArm);
    ang[a2++, f] = flex(uaL); ang[a2++, f] = flex(uaR); ang[a2++, f] = abd(uaL, -1); ang[a2++, f] = abd(uaR, 1);
    var trunk = P(HumanBodyBones.Neck) - P(HumanBodyBones.Hips);
    ang[a2++, f] = Mathf.Atan2(Vector3.Dot(trunk, fwd), Vector3.Dot(trunk, up)) * Mathf.Rad2Deg;
    ang[a2++, f] = Mathf.Atan2(Vector3.Dot(trunk, right), Vector3.Dot(trunk, up)) * Mathf.Rad2Deg;
    var hipLine = P(HumanBodyBones.RightUpperLeg) - P(HumanBodyBones.LeftUpperLeg); var shLine = P(HumanBodyBones.RightUpperArm) - P(HumanBodyBones.LeftUpperArm);
    float py = yaw(hipLine), cy = yaw(shLine);
    ang[a2++, f] = py; ang[a2++, f] = cy; ang[a2++, f] = Mathf.DeltaAngle(py, cy);
    ang[a2++, f] = Mathf.Atan2(Vector3.Dot(hipLine, up), Vector3.ProjectOnPlane(hipLine, up).magnitude) * Mathf.Rad2Deg;
    var face = wd[HumanBodyBones.Head] * fwd;
    ang[a2++, f] = Mathf.Atan2(-Vector3.Dot(face, up), Vector3.ProjectOnPlane(face, up).magnitude) * Mathf.Rad2Deg;
    ang[a2++, f] = Vector3.SignedAngle(fwd, Vector3.ProjectOnPlane(face, up), up);
    ang[a2++, f] = P(HumanBodyBones.LeftFoot).y - restPos[HumanBodyBones.LeftFoot].y; ang[a2++, f] = P(HumanBodyBones.RightFoot).y - restPos[HumanBodyBones.RightFoot].y;
    ang[a2++, f] = tf.ContainsKey(HumanBodyBones.LeftToes) ? P(HumanBodyBones.LeftToes).y - restPos[HumanBodyBones.LeftToes].y : float.NaN;
    ang[a2++, f] = tf.ContainsKey(HumanBodyBones.RightToes) ? P(HumanBodyBones.RightToes).y - restPos[HumanBodyBones.RightToes].y : float.NaN;
    var feet = P(HumanBodyBones.RightFoot) - P(HumanBodyBones.LeftFoot);
    ang[a2++, f] = Vector3.Dot(feet, right); ang[a2++, f] = Vector3.Dot(feet, fwd);
    ang[a2++, f] = P(HumanBodyBones.LeftHand).y - P(HumanBodyBones.Hips).y; ang[a2++, f] = P(HumanBodyBones.RightHand).y - P(HumanBodyBones.Hips).y;
  }
  // Root 移動軌跡：另開一隻開 Root Motion 的，從頭連續播
  var rm = spawn(clip, true); var rgraph = lastGraph[0];
  rgraph.Evaluate(0f); rm.transform.SetPositionAndRotation(Vector3.zero, Quaternion.identity);
  var rootS = new System.Text.StringBuilder();
  for (int si = 0; si <= S; si++) {
    if (si > 0) rgraph.Evaluate(1f / (fr * K));
    rootS.Append(si == 0 ? "" : ",").Append("[").Append(F4(-rm.transform.position.x)).Append(",").Append(F4(rm.transform.position.y)).Append(",").Append(F4(rm.transform.position.z)).Append(",").Append(F4(-rm.transform.eulerAngles.y)).Append("]");
  }
  // ---------- angles.json ----------
  var aj = new System.Text.StringBuilder();
  aj.Append("{\n\"format\":\"vrm-normalized-pose/2\",\"id\":").Append(J(motionId)).Append(",\"character\":").Append(J(src.name)).Append(",\"frameRate\":").Append(F(fr)).Append(",\"frameCount\":").Append(N + 1)
    .Append(",\"samplesPerFrame\":").Append(K).Append(",\"sampleCount\":").Append(S + 1)
    .Append(",\"loop\":").Append(st.loopTime ? "true" : "false").Append(",\"footIK\":").Append(footIK ? "true" : "false")
    .Append(",\n\"coordinate\":\"glTF／three.js 右手座標，+Y 朝上，角色面向 +Z，公尺；Unity 數值換算：位置 (-x,y,z)、旋轉 (x,-y,-z,w)\"")
    .Append(",\n\"rotationSpace\":\"VRM normalized：每根骨頭相對 T-pose 的旋轉（T-pose 時全部是單位四元數），local 是相對父骨頭\"")
    .Append(",\n\"sampling\":\"取自 Unity 實際播放原檔的結果；每格切 samplesPerFrame 份，第 i 份的時間＝i ÷ (frameRate × samplesPerFrame) 秒\"")
    .Append(",\n\"bones\":[");
  for (int i = 0; i < bones.Count; i++) aj.Append(i == 0 ? "" : ",").Append("{\"name\":").Append(J(vrmName(bones[i]))).Append(",\"parent\":").Append(parent[bones[i]] < 0 ? "null" : J(vrmName((HumanBodyBones)parent[bones[i]]))).Append("}");
  aj.Append("],\n\"localRotations\":{");
  for (int i = 0; i < bones.Count; i++) aj.Append(i == 0 ? "\n" : ",\n").Append(J(vrmName(bones[i]))).Append(":[").Append(rot[bones[i]]).Append("]");
  aj.Append("},\n\"hipsPosition\":[").Append(hipsPos).Append("],\n\"rootMotion\":[").Append(rootS).Append("],\n\"rootMotionFields\":[\"x\",\"y\",\"z\",\"yawDeg\"]")
    .Append(",\n\"worldPositions\":{");
  for (int i = 0; i < bones.Count; i++) aj.Append(i == 0 ? "\n" : ",\n").Append(J(vrmName(bones[i]))).Append(":[").Append(pos[bones[i]]).Append("]");
  aj.Append("},\n\"weapons\":[").Append(string.Join(",", wSpec.ToArray())).Append("]");

  var an = new System.Text.StringBuilder();
